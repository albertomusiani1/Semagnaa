difficolta: facile

data: 2026-01-30
ingredienti:
  - nome: Finocchi
    quantita: 2
    unita: pezzi
    reparto: frutta-verdura
  - nome: Arance
    quantita: 2
    unita: pezzi
    reparto: frutta-verdura
    note: non troppo dolci
  - nome: Olio extravergine di oliva
    quantita: 2
    unita: cucchiai
    reparto: dispensa
  - nome: Sale
    unita: qb
    reparto: dispensa
passaggi:
  - testo: Pulisci i finocchi togliendo la base e le foglie esterne, poi affettali il più sottile possibile.
  - testo: Pela le arance a vivo raccogliendo il succo che cade e tagliale a fette.
  - testo: Condisci con il succo raccolto, l'olio e il sale, mescola e servi subito.
